cmake_minimum_required(VERSION 3.20)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/version.json" SAAMAN_METADATA)
string(JSON SAAMAN_VERSION GET "${SAAMAN_METADATA}" version)
project(SAAMAN VERSION ${SAAMAN_VERSION} LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "EuroScope requires an x86 Windows build. Configure with -A Win32.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(WEBVIEW2_SDK "${CMAKE_SOURCE_DIR}/external/webview2" CACHE PATH "Extracted Microsoft.Web.WebView2 NuGet package")
if(NOT EXISTS "${WEBVIEW2_SDK}/build/native/include/WebView2.h")
  message(FATAL_ERROR "Run scripts/setup-sdk.ps1 or supply -DWEBVIEW2_SDK=<SDK directory>.")
endif()
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/generated/assets.rc")
  message(FATAL_ERROR "Build ui and run scripts/embed-assets.mjs first.")
endif()
add_library(saaman_host STATIC native/context.cpp native/host.cpp native/auth_client.cpp native/coordination_popup.cpp)
target_include_directories(saaman_host PUBLIC native external/include generated "${WEBVIEW2_SDK}/build/native/include")
target_compile_definitions(saaman_host PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(saaman_host PUBLIC "${WEBVIEW2_SDK}/build/native/x86/WebView2LoaderStatic.lib" ole32 oleaut32 user32 gdi32 shlwapi shell32 version advapi32 winhttp crypt32)
add_library(SAAMAN SHARED native/plugin.cpp native/telemetry.cpp native/radar_screen.cpp generated/assets.rc)
target_link_libraries(SAAMAN PRIVATE saaman_host "${CMAKE_SOURCE_DIR}/external/lib/EuroScopePlugInDLL.lib")
target_compile_definitions(SAAMAN PRIVATE SAAMAN_VERSION="${PROJECT_VERSION}")
add_executable(saaman_host_test tests/host-test.cpp generated/assets.rc)
target_link_libraries(saaman_host_test PRIVATE saaman_host)
add_executable(saaman_context_test tests/context-test.cpp)
target_link_libraries(saaman_context_test PRIVATE saaman_host)
enable_testing()
add_test(NAME saaman-context COMMAND saaman_context_test "${CMAKE_BINARY_DIR}/context-test-data")
